SET ProjectDir TO $'''C:\\Users\\hello\\nexvia-technologies-internship-projects\\DA-04'''
SET ReportMonth TO $'''2018-12'''
SET PdfPath TO $'''%ProjectDir%\\reports\\Arcadia_Board_Report_%ReportMonth%.pdf'''

Scripting.RunDOSCommand.RunDOSCommand DOSCommandOrApplication: $'''\"%ProjectDir%\\.venv\\Scripts\\python.exe\" src\\run_pipeline.py --skip-build''' WorkingDirectory: ProjectDir StandardOutput=> CommandOutput StandardError=> CommandErrorOutput ExitCode=> CommandExitCode
IF CommandExitCode <> 0 THEN
    Email.SendEmail.AuthenticateAndSend SMTPServer: $'''smtp.gmail.com''' Port: 587 EnableSSL: True Username: $'''your.sender@gmail.com''' Password: $'''<YOUR_GMAIL_APP_PASSWORD>''' AcceptUntrustedCertificates: False SendFrom: $'''your.sender@gmail.com''' SendTo: RecipientEmail Subject: $'''FAILED: Arcadia board report – data refresh (%ReportMonth%)''' Body: $'''The monthly data refresh failed, so no board report was sent.

Pipeline output: %CommandOutput%

Errors: %CommandErrorOutput%''' IsBodyHtml: False
    EXIT Code: 0 ErrorMessage: $'''Data refresh failed - see email'''
END
Scripting.RunDOSCommand.RunDOSCommand DOSCommandOrApplication: $'''cscript //nologo \"%ProjectDir%\\automation\\refresh_and_export.vbs\" \"%PdfPath%\"''' WorkingDirectory: ProjectDir StandardOutput=> CommandOutput2 StandardError=> CommandErrorOutput2 ExitCode=> CommandExitCode2

    Email.SendEmail.AuthenticateAndSend SMTPServer: $'''smtp.gmail.com''' Port: 587 EnableSSL: True Username: $'''your.sender@gmail.com''' Password: $'''<YOUR_GMAIL_APP_PASSWORD>''' AcceptUntrustedCertificates: False SendFrom: $'''your.sender@gmail.com''' SendTo: RecipientEmail Subject: $'''FAILED: Arcadia board report – PDF export (%ReportMonth%)''' Body: $'''The board report PDF could not be created.

    EXIT Code: 0 ErrorMessage: $'''PDF export failed - see email'''
END
