// =====================================================================
// Power Query (M) code for the DA-04 Power Pivot model
// One query per CSV in excel/data/. All queries read the folder from ONE
// parameter, DataFolder, so the workbook works on any computer: change the
// parameter, Refresh All, done.
//
// HOW TO USE (Excel -> Data -> Get Data -> Launch Power Query Editor):
//   1. Home -> Manage Parameters -> New Parameter
//        Name: DataFolder   Type: Text
//        Current Value: the full path of your excel\data folder, ENDING WITH A BACKSLASH, e.g.
//        C:\Users\hello\nexvia-technologies-internship-projects\DA-04\excel\data\
//   2. For each block below: Home -> New Source -> Other Sources -> Blank Query,
//      then Home -> Advanced Editor, replace everything with the block's code, Done,
//      and rename the query (right pane, Name) to the name in the block header.
//   3. Home -> Close & Load -> Close & Load To... -> "Only Create Connection"
//      AND tick "Add this data to the Data Model" -> OK.
// =====================================================================

// ---------------------------------------------------------------------
// Query name: dim_month        (36 rows)
// ---------------------------------------------------------------------
let
    Source = Csv.Document(File.Contents(DataFolder & "dim_month.csv"),
                          [Delimiter = ",", Encoding = 65001, QuoteStyle = QuoteStyle.Csv]),
    Headers = Table.PromoteHeaders(Source, [PromoteAllScalars = true]),
    // "en-US" culture: dates are YYYY-MM-DD and decimals use "." whatever the PC's region is
    Typed = Table.TransformColumnTypes(Headers, {
        {"month_start", type date},
        {"month_end", type date},
        {"year_num", Int64.Type},
        {"quarter_num", Int64.Type},
        {"month_num", Int64.Type},
        {"month_name", type text},
        {"year_month_label", type text},
        {"fiscal_year", Int64.Type}
    }, "en-US")
in
    Typed
